\section{Introducción y objetivos}
% ============================================================================

\begin{frame}{Motivación}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{El problema}
        Cada sesión bursátil un inversor decide \textbf{comprar, vender o
        mantener} un activo disponiendo únicamente de información histórica
        y una estimación del futuro.
      \end{block}
      \vspace{0.4cm}
      Este es un \textbf{proceso de decisión secuencial bajo incertidumbre}:
      \begin{itemize}
        \item El estado futuro es desconocido.
        \item Se pueden asignar \emph{probabilidades} a sus evoluciones.
        \item Se busca \emph{maximizar el valor esperado} de un objetivo.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{alertblock}{Pregunta guía}
        ¿Existe una política $\pi:\mathcal{S}\to\mathcal{A}$ que obtenga
        sistemáticamente un retorno superior al de la estrategia pasiva
        \emph{Buy \& Hold}?
      \end{alertblock}
      \vspace{0.3cm}
      \begin{exampleblock}{Herramientas}
        \begin{itemize}
          \item Variables aleatorias
          \item Simulación estocástica
          \item Estimación por media muestral
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Conexión con el Monte Carlo clásico}
  \begin{block}{Idea central}
    El MCTS es una \textbf{\textcolor{orange}{aplicación directa}} del estimador de Monte Carlo por media muestral.
  \end{block}

  \vspace{0.4cm}
  Si $X_i$ es el retorno al simular una trayectoria de precios durante $H$
  días tomando la acción $a$, repitiéndolo $n$ veces de forma independiente:

  \[
    \widehat{I}_n = \frac{1}{n}\sum_{i=1}^{n} X_i
    \;\xrightarrow{\ \text{c.s.}\ }\;
    \mathbb{E}[X \mid a]
    \qquad (n\to\infty) 
  \]
  por la \textbf{Ley Fuerte de los Grandes Números}.

  \vspace{0.4cm}
  \begin{alertblock}{}
  En cada nodo del MCTS:
  \begin{equation*}
  \frac{Q(a)}{N(a)}\Leftrightarrow \widehat{I}_n
  \end{equation*}
    aplicado a los rollouts ejecutados hasta ese momento.
  \end{alertblock}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Objetivos del trabajo}
  \begin{enumerate}
    \item[\textbf{O1}] \textbf{Formalizar} el problema de trading como un
      Proceso de Decisión de Markov (MDP) con espacio de acciones discreto.
    \vspace{0.15cm}
    \item[\textbf{O2}] \textbf{Implementar} el algoritmo MCTS con política UCB1
      y rollouts generados mediante Movimiento Browniano Geométrico (GBM)
      calibrado sobre ventana móvil.
    \vspace{0.15cm}
    \item[\textbf{O3}] \textbf{Demostrar} que la convergencia se apoya en la
      LLN y en la cota de error $\mathcal{O}(n^{-1/2})$.
    \vspace{0.15cm}
    \item[\textbf{O4}] \textbf{Comparar} MCTS con Buy \& Hold mediante ROI,
      Sharpe, alfa y drawdown.
    \vspace{0.15cm}
    \item[\textbf{O5}] \textbf{Validar} experimentalmente la cota
      $\mathcal{O}(n^{-1/2})$ observando la convergencia del árbol.
  \end{enumerate}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Monte Carlo por media muestral}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{Estimador}
        Sea $X$ v.a.\ con $\mathbb{E}[X] = \mu$ finita. Dadas $X_1,\ldots,X_n$
        i.i.d.:
        \[
          \widehat{I}_n = \frac{v}{n}\sum_{i=1}^n X_i
        \]
        donde $v$ es el volumen del dominio de integración.
      \end{block}
      \vspace{0.3cm}
      \begin{block}{Ley Fuerte de los Grandes Números}
        \[
          \widehat{I}_n \xrightarrow{\ \text{c.s.}\ } v\cdot\mu
          \qquad (n\to\infty)
        \]
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{exampleblock}{Cota de error}
        Por el TCL, para $n$ suficientemente grande:
        \[
          \varepsilon_n
          = \frac{v\,\sigma}{\sqrt{n}}\,z_{\alpha/2}
          = \mathcal{O}(n^{-1/2})
        \]
        donde $\sigma^2 = \mathrm{Var}(X)$.
      \end{exampleblock}
      \vspace{0.3cm}
      \begin{alertblock}{Implicación para MCTS}
        \textbf{Doblar la precisión} requiere \textbf{cuadruplicar}
        las iteraciones. Justifica elegir\\
        \centering
        \texttt{ITERACIONES} $= 5.000$.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Proceso de Decisión de Markov (MDP)}
  \begin{block}{Definición}
    Un MDP es una cuádrupla $\mathcal{M} = (\mathcal{S},\mathcal{A},P,R)$:
    \begin{itemize}
      \item $\mathcal{S}$: espacio de \textbf{estados}
      \item $\mathcal{A}$: espacio de \textbf{acciones}
      \item $P(s'\mid s,a)$: \textbf{núcleo de transición}
        (distribución del estado siguiente)
      \item $R:\mathcal{S}\times\mathcal{A}\to\mathbb{R}$:
        \textbf{función de recompensa}
    \end{itemize}
    La \textbf{propiedad de Markov}: $S_{t+1}\perp$ pasado $\mid (S_t,A_t)$.
  \end{block}

  %\vspace{0.3cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Política y función de valor:}
      \[
        V^\pi(s) = \mathbb{E}_\pi\!\left[
          \sum_{t\geq 0}\gamma^t R(S_t,A_t)\,\Big|\,S_0=s
        \right]
      \]
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Ecuación de Bellman:}
      \[
        V^*(s) = \max_{a\in\mathcal{A}}\!\left\{
          R(s,a) + \gamma\sum_{s'} P(s'|s,a)\,V^*(s')
        \right\}
      \]
    \end{column}
  \end{columns}

  %\vspace{0.2cm}
  \begin{alertblock}{}
    Resolver Bellman directamente requiere \emph{enumerar} $\mathcal{S}$
    $\to$ inviable en trading. MCTS lo esquiva mediante simulación estocástica.
  \end{alertblock}
\end{frame}

\begin{frame}{UCB1: dilema exploración–explotación}
  \begin{block}{Fórmula UCB1 \footnotesize{(Kocsis \& Szepesvári, 2006)}}
    \[
      a^*_{\text{UCB1}}(s)
      \;=\;
      \arg\max_{a\in\mathcal{A}(s)}
      \Bigl\{
        \underbrace{\dfrac{Q(s,a)}{N(s,a)}}_{\text{explotación}}
        \;+\;
        \sqrt{2}\,\underbrace{\sqrt{\dfrac{\ln N(s)}{N(s,a)}}}_{\text{exploración}}
      \Bigr\}\hspace{2em}
    \]
  \end{block}

  \vspace{0.3cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Término de explotación}
        Favorece acciones con \textbf{alta recompensa media} ya observada.
      \end{exampleblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Término de exploración}
        Favorece acciones \textbf{poco visitadas}: si $N(s,a)$ se mantiene
        acotado mientras $N(s)\to\infty$, el término diverge
        $\to$ toda acción se visita infinitas veces.
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.3cm}
  MCTS con UCB1 es un método iterativo tipo \textbf{Gauss–Seidel adaptativo}:
  actualiza una sola componente $q^{(k)}(a)$ por iteración,
  eligiendo la componente mediante la fórmula UCB1 en lugar de un orden cíclico.
\end{frame}

\begin{frame}{Movimiento Browniano Geométrico (GBM)}
  \textbf{Motor de simulación del rollout.}
  Bajo la hipótesis de log-retornos gaussianos el precio satisface:
  \[
    dS_t = \mu S_t\,dt + \sigma S_t\,dW_t
  \]

  \begin{block}{Solución exacta (Lema de Itô)}
    \[
      S_{t+h} = S_t\exp\!\left(
        \Bigl(\mu - \tfrac{\sigma^2}{2}\Bigr)h
        + \sigma\sqrt{h}\,Z
      \right), \qquad Z\sim\mathcal{N}(0,1)
    \]
    Sin error de discretización (solución analítica, no Euler–Maruyama).
  \end{block}
\end{frame}

% ---------------------------------------------------------------------------

\begin{frame}
      \begin{exampleblock}{Calibración rolling}
        $(\hat\mu,\hat\sigma)$ estimados sobre los últimos
        \texttt{VENTANA\_CALIB}$=60$ días de log-retornos:
        \[
          r_k = \ln\!\tfrac{P_k}{P_{k-1}}
        \]
        \textbf{Sin lookforward bias}: solo se usan datos pasados.
      \end{exampleblock}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Convergencia del MCTS}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{block}{Argumento}
        Para cada acción $a\in\mathcal{A}$, UCB1 garantiza
        $N(s_0,a)\to\infty$ conforme el total de iteraciones $n\to\infty$.
        Por el Corolario de la LLN:
        \[
          \frac{Q(s_0,a)}{N(s_0,a)}
          \;\xrightarrow{\text{c.s.}}\;
          \mathbb{E}[R_T\mid s_0,a]
        \]
      \end{block}

      \vspace{0.1cm}
      \begin{block}{Error del estimador}
        Recurrencia lineal con $\alpha_k = 1/(N^{(k)}(a)+1)$:
        \[
          q^{(k+1)}(a) = (1-\alpha_k)\,q^{(k)}(a) + \alpha_k X_{k+1}
        \]
        La varianza decae: $\mathrm{Var}(e^{(k)}) = \sigma^2(a)/N^{(k)}(a) = \mathcal{O}(N^{-1})$,
        luego error $= \mathcal{O}(N^{-1/2})$.
      \end{block}
    \end{column}
    \begin{column}{0.42\textwidth}
      \textbf{MCTS como método iterativo:}
      \vspace{0.15cm}

      \begin{center}
      \small
      \begin{tabular}{@{}p{2.8cm}p{2.8cm}@{}}
        \toprule
        \textbf{Tema 6} & \textbf{MCTS} \\
        \midrule
        Iterado $x^{(k)}$ & $q^{(k)}(a)$ \\
        Solución $x^*$ & $\mathbb{E}[R_T\mid s_0,a]$ \\
        Término indep.\ $b$ & Retorno $X_{k+1}$ \\
        Matriz iteración $Q$ & $1-\alpha_k$ \\
        $\rho(Q)<1$ & $|1-\alpha_k|<1$ \checkmark \\
        Gauss–Seidel & MCTS con UCB1 \\
        Relajación $\omega$ & Exploración $c$ \\
        \bottomrule
      \end{tabular}
      \end{center}
    \end{column}
  \end{columns}
\end{frame}

%    Recompensa $=$ exceso de log-retorno del agente sobre B\&H.

%  \end{block}
%
%  \vspace{0.08cm}
  \begin{block}{\emph{Common Random Numbers} en el rollout}
    Agente y B\&H usan el \textbf{mismo} $Z\sim\mathcal{N}(0,1)$:
    misma trayectoria de mercado, solo cambia la decisión.
    Reduce la varianza de la estimación comparativa.
  \end{block}

  \vspace{0.08cm}
  \begin{block}{Política de selección final: max-visits}
    Se elige $\arg\max_a N(s_0,a)$ en lugar de $\arg\max_a Q/N$.
    Con $Q/N$ alto pero $N$ pequeño el error estándar $\sigma/\sqrt{N}$ es grande
    $\to$ max-visits es estadísticamente más robusta.
  \end{block}
\end{frame}

\begin{frame}{Limitaciones del modelo}
  \vspace{0.3cm}
  \centering
  \begin{tikzpicture}[
    card/.style={rectangle, rounded corners=6pt,
                 fill=uvblue!7, draw=uvblue!30, line width=0.8pt,
                 minimum width=13.5cm, text width=12.2cm,
                 inner sep=10pt},
    badge/.style={circle, fill=uvblue, text=white,
                  font=\bfseries\large,
                  minimum size=0.85cm, inner sep=0pt}
  ]
    % --- Tarjeta L1 ---
    \node[card] (c1) at (0, 0) {%
      \hspace{1.1cm}%
      \parbox{11cm}{\small
        \textbf{GBM con colas ligeras}\\[3pt]
        Los log-retornos gaussianos subestiman eventos extremos (\emph{fat tails}).
        La caída $-75\%$ de TSLA ocurre en la realidad con mucha más frecuencia
        de la que predice el modelo.
        Extensión natural: modelo de Merton (GBM + saltos de Poisson).
      }
    };
    \node[badge] at ([xshift=-5.8cm]c1.center) {L1};

    % --- Tarjeta L2 (anclada bajo c1 para alineación exacta) ---
    \node[card, below=0.45cm of c1] (c2) {%
      \hspace{1.1cm}%
      \parbox{11cm}{\small
        \textbf{Validez limitada al periodo analizado}\\[3pt]
        Los hiperparámetros no se han separado en \emph{in-sample / out-of-sample},
        y TSLA en este periodo es especialmente favorable para \emph{market timing}.
        Una evaluación rigurosa requiere validación cruzada temporal
        y repetición con distintas semillas y activos.
      }
    };
    \node[badge] at ([xshift=-5.8cm]c2.center) {L2};
  \end{tikzpicture}
\end{frame}
